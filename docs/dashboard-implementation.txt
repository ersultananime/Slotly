2. Примените supabase/migrations/0002_dashboard.sql к существующей схеме из задания или после 0001. Миграция не применена автоматически.
3. Если у компании несколько записей knowledge_base, объедините их перед миграцией: миграция остановится без удаления данных.
4. Старая альтернативная supabase_slotly_schema.sql с client_name/phone/content_type не соответствует схеме задания и не поддерживается этой миграцией. Не запускайте её для этого изменения.

Архитектура:
Пользователь проверяется через auth.getUser() на сервере. company_id определяется на сервере перед чтением и сохранением. Клиентская форма получает предзаполненный текст и вызывает Server Action. Proxy обновляет cookies и перенаправляет неавторизованные запросы на /login.

Порядок файлов: общая инфраструктура, кабинет и шаг 1, шаг 2, шаг 3, состояния загрузки/ошибки, миграция.

- Сохраните текст компании A, обновите страницу: текст должен сохраниться; повторное сохранение не создаёт вторую запись.
- Аккаунт B не видит текст и лиды компании A. Запросы к Supabase от B с company_id компании A блокируются RLS.
- Код виджета содержит UUID текущей компании; после копирования галочка исчезает через 2 секунды.

import 'server-only'
import { createServerClient as createSSRClient } from '@supabase/ssr'
import { cookies } from 'next/headers'

export async function createServerClient() {
  const store = await cookies()
  return createSSRClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!, {
    cookies: {
      getAll: () => store.getAll(),
      setAll(values) {
        try { values.forEach(({ name, value, options }) => store.set(name, value, options)) }
        catch { /* Server Components cannot set cookies; proxy refreshes them. */ }
      },
    },
  })
}

export const requireUser = cache(async () => {
  const supabase = await createServerClient()
  const { data: { user }, error } = await supabase.auth.getUser()
  if (error || !user) redirect('/login')
  return { supabase, user }
})

export const requireCompany = cache(async () => {
  const { supabase, user } = await requireUser()
  const { data: company, error } = await supabase.from('companies')
    .select('id').eq('user_id', user.id).maybeSingle()
  if (error) throw new Error('Не удалось загрузить компанию.')
  if (!company) throw new Error('Компания не найдена. Обратитесь в поддержку Slotly.')
  return { supabase, user, companyId: company.id as string }
})

import { createServerClient } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

export async function proxy(request: NextRequest) {
  let response = NextResponse.next({ request })
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    { cookies: {
      getAll: () => request.cookies.getAll(),
      setAll(values) {
        values.forEach(({ name, value }) => request.cookies.set(name, value))
        response = NextResponse.next({ request })
        values.forEach(({ name, value, options }) => response.cookies.set(name, value, options))
      },
    } },
  )
  const { data: { user }, error } = await supabase.auth.getUser()
  if ((error || !user) && request.nextUrl.pathname.startsWith('/dashboard')) {
    const url = request.nextUrl.clone()
    url.pathname = '/login'
    url.search = ''
    const redirect = NextResponse.redirect(url)
    response.cookies.getAll().forEach(cookie => redirect.cookies.set(cookie))
    response = redirect
  }
  response.headers.set('Cache-Control', 'private, no-store')
  return response
}

export const config = { matcher: ['/dashboard/:path*', '/login', '/register'] }

'use server'

import { revalidatePath } from 'next/cache'
import { requireCompany } from '@/lib/dashboard'

export async function saveKnowledge(content: string) {
  // Recheck ownership for each mutation; company_id is never accepted from the browser.
  const { supabase, companyId } = await requireCompany()
  if (typeof content !== 'string' || !content.trim() || content.length > 50000) {
    return { error: 'Введите от 1 до 50 000 символов.' }
  }
  const { data, error } = await supabase.from('knowledge_base')
    .upsert({ company_id: companyId, content: content.trim() }, { onConflict: 'company_id' })
    .select('id').single()
  if (error || !data) return { error: 'Не удалось сохранить базу знаний. Попробуйте ещё раз.' }
  revalidatePath('/dashboard')
  return { success: true }
}

        setContent(analysis.content)
        setShowEditor(true)
        setAnalysis(null)
        setNotice({})
      }} className="mt-4 rounded-xl bg-blue-600 px-5 py-3 text-sm font-bold text-white hover:bg-blue-700 disabled:opacity-60">{content.trim() ? 'Заменить текст в редакторе' : 'Использовать это описание'}</button>
      <button type="button" disabled={busy} onClick={() => setAnalysis(null)} className="ml-3 mt-3 rounded-xl px-4 py-3 text-sm font-semibold text-slate-600">Отменить</button>
    </section>}

    {!showEditor && <button type="button" disabled={busy} onClick={() => setShowEditor(true)} className="text-sm font-semibold text-blue-700 underline underline-offset-4 disabled:opacity-60">Нет сайта? Заполнить вручную</button>}

    {showEditor && <form onSubmit={event => {
      event.preventDefault()
      setNotice({})
      startTransition(async () => {
        try {
          const result = await saveKnowledge(content)
          setNotice(result)
          if (result.success) {
            window.location.hash = 'widget-code'
            document.getElementById('widget-code')?.focus({ preventScroll: true })
          }
        } catch (error) {
          unstable_rethrow(error)
          setNotice({ error: 'Не удалось сохранить данные. Проверьте соединение и повторите попытку.' })
        }
      })
    }} className="space-y-4" aria-busy={isLoading}>
      <label htmlFor="knowledge-content" className="block text-sm font-semibold text-slate-700">Информация для консультанта</label>
      <p id="knowledge-hint" className="text-sm text-slate-500">Уточните услуги, цены и ответы на частые вопросы. Консультант будет использовать сохранённый текст.</p>
      <textarea id="knowledge-content" name="content" rows={12} required maxLength={50000}
        aria-describedby="knowledge-hint" value={content} disabled={busy}
        onChange={event => { setContent(event.target.value); setNotice({}) }}
        className="w-full resize-y rounded-xl border border-slate-300 bg-white p-4 text-base text-slate-900 focus:border-blue-500 focus:ring-2 focus:ring-blue-200 disabled:opacity-60" />
      <button type="submit" disabled={busy || !content.trim()} className="inline-flex items-center gap-2 rounded-xl bg-blue-600 px-5 py-3 text-sm font-bold text-white hover:bg-blue-700 disabled:cursor-not-allowed disabled:opacity-60">
        {isLoading ? 'Сохраняем…' : 'Сохранить и получить код виджета'}<ArrowRight size={18} aria-hidden="true" />
      </button>
      <div aria-live="polite" role="status" className="text-sm">
        {notice.error && <p className="text-red-700">{notice.error}</p>}
        {notice.success && <p className="text-emerald-700">База знаний сохранена. Скопируйте код виджета ниже.</p>}
      </div>
    </form>}
  </div>
}

'use client'

import { useEffect, useRef, useState } from 'react'
import { Check, Copy } from 'lucide-react'

export default function WidgetCode({ companyId }: { companyId: string }) {
  const [copied, setCopied] = useState(false)
  const [error, setError] = useState('')
  const timeout = useRef<ReturnType<typeof setTimeout> | null>(null)
  useEffect(() => () => { if (timeout.current) clearTimeout(timeout.current) }, [])
  const code = `<script src="https://slotly-inky.vercel.app/widget.js" data-company-id="${companyId}"></script>`
  async function copyCode() {
    setError('')
    try {
      await navigator.clipboard.writeText(code)
      setCopied(true)
      if (timeout.current) clearTimeout(timeout.current)
      timeout.current = setTimeout(() => setCopied(false), 2000)
    } catch {
      setCopied(false)
      setError('Не удалось скопировать код. Выделите его и скопируйте вручную.')
    }
  }
  return <div className="mt-5 space-y-4">
    <p className="text-sm text-slate-500">Добавьте код перед закрывающим тегом &lt;/body&gt; на вашем сайте.</p>
    <pre tabIndex={0} aria-label="Код установки виджета" className="overflow-x-auto rounded-xl bg-slate-950 p-5 text-sm leading-7 text-emerald-300"><code>{code}</code></pre>
    <button type="button" onClick={copyCode} className="inline-flex items-center gap-2 rounded-xl border border-slate-300 px-4 py-3 text-sm font-bold hover:bg-slate-50">
      {copied ? <Check size={18} aria-hidden="true" /> : <Copy size={18} aria-hidden="true" />}
      {copied ? 'Скопировано' : 'Скопировать код'}
    </button>
    <p role="status" aria-live="polite" className="text-sm text-slate-700">{error || (copied ? 'Код скопирован в буфер обмена.' : '')}</p>
  </div>
}

-- Applies to the schema described for the dashboard, or after 0001.
-- Does not drop tables or silently discard duplicate knowledge records.
BEGIN;

-- Compatibility with 0001, which called the message column "summary".
DO $$ BEGIN
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'leads' AND column_name = 'message') THEN
    ALTER TABLE public.leads ADD COLUMN message text;
    IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'leads' AND column_name = 'summary') THEN
      UPDATE public.leads SET message = summary;
    END IF;
  END IF;
  IF EXISTS (SELECT 1 FROM public.knowledge_base GROUP BY company_id HAVING count(*) > 1) THEN
    RAISE EXCEPTION 'Multiple knowledge records per company: merge them before applying 0002_dashboard.sql';
  END IF;
END $$;

CREATE UNIQUE INDEX IF NOT EXISTS dashboard_knowledge_company_unique ON public.knowledge_base(company_id);
CREATE INDEX IF NOT EXISTS dashboard_leads_company_created ON public.leads(company_id, created_at DESC, id DESC);

ALTER TABLE public.companies ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.knowledge_base ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.leads ENABLE ROW LEVEL SECURITY;

CREATE POLICY dashboard_company_read ON public.companies FOR SELECT TO authenticated USING (user_id = auth.uid());
CREATE POLICY dashboard_knowledge_read ON public.knowledge_base FOR SELECT TO authenticated USING (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));
CREATE POLICY dashboard_knowledge_insert ON public.knowledge_base FOR INSERT TO authenticated WITH CHECK (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));
CREATE POLICY dashboard_knowledge_update ON public.knowledge_base FOR UPDATE TO authenticated USING (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid())) WITH CHECK (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));
CREATE POLICY dashboard_leads_read ON public.leads FOR SELECT TO authenticated USING (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));

-- Restrictive policies also constrain any pre-existing permissive owner policies.
CREATE POLICY dashboard_company_boundary ON public.companies AS RESTRICTIVE FOR ALL TO authenticated USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());
CREATE POLICY dashboard_knowledge_boundary ON public.knowledge_base AS RESTRICTIVE FOR ALL TO authenticated USING (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid())) WITH CHECK (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));
CREATE POLICY dashboard_leads_boundary ON public.leads AS RESTRICTIVE FOR ALL TO authenticated USING (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid())) WITH CHECK (company_id IN (SELECT id FROM public.companies WHERE user_id = auth.uid()));
COMMIT;
